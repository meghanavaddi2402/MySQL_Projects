mysql> -- VIEW: Live league standings
mysql> -- ============================================================
mysql> 
mysql> CREATE VIEW Standings AS
    -> WITH team_matches AS (
    ->     SELECT HomeTeamID AS TeamID, HomeScore AS GoalsFor, AwayScore AS GoalsAgainst,
    ->            CASE WHEN HomeScore > AwayScore THEN 'W'
    ->                 WHEN HomeScore < AwayScore THEN 'L'
    ->                 ELSE 'D' END AS Result
    ->     FROM Match_ WHERE Status = 'Completed'
    ->     UNION ALL
    ->     SELECT AwayTeamID AS TeamID, AwayScore AS GoalsFor, HomeScore AS GoalsAgainst,
    ->            CASE WHEN AwayScore > HomeScore THEN 'W'
    ->                 WHEN AwayScore < HomeScore THEN 'L'
    ->                 ELSE 'D' END AS Result
    ->     FROM Match_ WHERE Status = 'Completed'
    -> )
    -> SELECT
    ->     t.TeamID,
    ->     t.TeamName,
    ->     COUNT(*) AS Played,
    ->     SUM(CASE WHEN tm.Result = 'W' THEN 1 ELSE 0 END) AS Wins,
    ->     SUM(CASE WHEN tm.Result = 'D' THEN 1 ELSE 0 END) AS Draws,
    ->     SUM(CASE WHEN tm.Result = 'L' THEN 1 ELSE 0 END) AS Losses,
    ->     SUM(tm.GoalsFor) AS GoalsFor,
    ->     SUM(tm.GoalsAgainst) AS GoalsAgainst,
    ->     SUM(tm.GoalsFor) - SUM(tm.GoalsAgainst) AS GoalDifference,
    ->     SUM(CASE WHEN tm.Result = 'W' THEN 3 WHEN tm.Result = 'D' THEN 1 ELSE 0 END) AS Points
    -> FROM team_matches tm
    -> JOIN Team t ON t.TeamID = tm.TeamID
    -> GROUP BY t.TeamID, t.TeamName
    -> ORDER BY Points DESC, GoalDifference DESC;

    ->        CASE
    ->            WHEN m.HomeTeamID = t.TeamID AND m.HomeScore > m.AwayScore THEN 'W'
